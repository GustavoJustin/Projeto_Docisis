// Integra o formulário de movimentação com a API do DoceGestão. document.addEventListener("DOMContentLoaded", function () { const formulario = document.getElementById("form-movimentacao"); const campoInsumo = document.getElementById("insumo"); const campoQuantidade = document.getElementById("quantidade"); const campoTipo = document.getElementById("tipo_movimentacao"); const corpoTabela = document.querySelector("main table tbody"); const campoBusca = document.querySelector('main input[placeholder="Buscar insumo..."]'); const token = localStorage.getItem("docegestao_token"); let insumos = []; let movimentacoes = []; if (!formulario || !campoInsumo || !campoQuantidade || !campoTipo || !corpoTabela) { return; } if (!token) { alert("Sua sessão expirou. Faça login novamente."); return; } // Todas as chamadas protegidas usam o token salvo no login. async function chamarApi(url, configuracao = {}) { const resposta = await fetch(url, { ...configuracao, headers: { Authorization: "Bearer " + token, ...(configuracao.body ? { "Content-Type": "application/json" } : {}), }, }); const resultado = await resposta.json(); if (!resposta.ok) { throw new Error(resultado.mensagem || "Não foi possível falar com a API."); } return resultado; } // O texto da opção identifica produto, marca e lote. function nomeCompletoDoInsumo(insumo) { let nome = insumo.tipo; if (insumo.marca) nome += " — " + insumo.marca; if (insumo.lote) nome += " — lote " + insumo.lote; return nome; } // Carrega os insumos cadastrados e mostra sugestões no campo existente. async function carregarInsumos() { const resultado = await chamarApi("/api/estoque/materias-primas"); insumos = resultado.dados || []; let lista = document.getElementById("lista-insumos-api"); if (!lista) { lista = document.createElement("datalist"); lista.id = "lista-insumos-api"; document.body.appendChild(lista); } campoInsumo.setAttribute("list", lista.id); lista.innerHTML = ""; insumos.forEach(function (insumo) { const opcao = document.createElement("option"); opcao.value = nomeCompletoDoInsumo(insumo); lista.appendChild(opcao); }); } // Adiciona uma célula ao histórico sem interpretar dados como HTML. function adicionarCelula(linha, texto, classes) { const celula = document.createElement("td"); celula.className = classes; celula.textContent = texto || "—"; linha.appendChild(celula); return celula; } // Mostra o histórico recebido da API na tabela que já existe na página. function mostrarHistorico(lista) { corpoTabela.innerHTML = ""; if (lista.length === 0) { const linha = document.createElement("tr"); const celula = document.createElement("td"); celula.colSpan = 5; celula.className = "py-4 text-center text-stone-400"; celula.textContent = "Nenhuma movimentação registrada ainda."; linha.appendChild(celula); corpoTabela.appendChild(linha); return; } lista.forEach(function (movimentacao, indice) { const linha = document.createElement("tr"); if (indice < lista.length - 1) linha.className = "border-b border-stone-50"; const data = new Date(movimentacao.data_hora); const dataFormatada = Number.isNaN(data.getTime()) ? movimentacao.data_hora : new Intl.DateTimeFormat("pt-BR", { dateStyle: "short", timeStyle: "short", }).format(data); adicionarCelula(linha, dataFormatada, "py-3 text-stone-600"); adicionarCelula(linha, movimentacao.insumo, "py-3 font-medium text-stone-800"); const celulaTipo = document.createElement("td"); celulaTipo.className = "py-3 text-center"; const etiqueta = document.createElement("span"); const entrada = movimentacao.tipo === "ENTRADA"; etiqueta.className = entrada ? "inline-flex rounded-full bg-emerald-50 px-2.5 py-1 text-xs font-semibold text-emerald-700" : "inline-flex rounded-full bg-rose-50 px-2.5 py-1 text-xs font-semibold text-rose-600"; etiqueta.textContent = entrada ? "ENTRADA" : "SAÍDA"; celulaTipo.appendChild(etiqueta); linha.appendChild(celulaTipo); const quantidade = Number(movimentacao.quantidade).toLocaleString("pt-BR"); const sinal = entrada ? "+ " : "− "; const corQuantidade = entrada ? "text-emerald-600" : "text-rose-600"; adicionarCelula( linha, sinal + quantidade, "py-3 text-right font-medium " + corQuantidade, ); adicionarCelula( linha, movimentacao.responsavel, "py-3 text-right text-stone-600", ); corpoTabela.appendChild(linha); }); } // Busca as movimentações e atualiza a tabela. async function carregarHistorico() { const resultado = await chamarApi("/api/estoque/movimentacoes"); movimentacoes = resultado.dados || []; mostrarHistorico(movimentacoes); } // Envia a movimentação quando o botão do formulário é clicado. formulario.addEventListener("submit", async function (evento) { evento.preventDefault(); const insumoEscolhido = insumos.find(function (insumo) { return nomeCompletoDoInsumo(insumo) === campoInsumo.value; }); if (!insumoEscolhido) { campoInsumo.setCustomValidity("Escolha um insumo da lista."); campoInsumo.reportValidity(); campoInsumo.addEventListener("input", function () { campoInsumo.setCustomValidity(""); }, { once: true }); return; } campoInsumo.setCustomValidity(""); const quantidade = Number(campoQuantidade.value); const botao = formulario.querySelector('button[type="submit"]'); if (botao) botao.disabled = true; try { const dadosFormulario = new FormData(formulario); const dadosMovimentacao = { id_produtos: Number(insumoEscolhido.id_produtos), tipo: campoTipo.value, quantidade: quantidade, motivo: dadosFormulario.get("motivo"), notas: dadosFormulario.get("notas"), }; await chamarApi("/api/estoque/movimentacoes", { method: "POST", body: JSON.stringify(dadosMovimentacao), }); formulario.reset(); if (typeof window.selecionarTipo === "function") { window.selecionarTipo("ENTRADA"); } await carregarInsumos(); await carregarHistorico(); alert("Movimentação registrada e estoque atualizado."); } catch (erro) { alert(erro.message); } finally { if (botao) botao.disabled = false; } }); // Filtra o histórico usando o campo de busca já existente na tela. if (campoBusca) { campoBusca.addEventListener("input", function () { const busca = campoBusca.value.toLocaleLowerCase("pt-BR"); const resultado = movimentacoes.filter(function (movimentacao) { const texto = [movimentacao.insumo, movimentacao.tipo, movimentacao.responsavel] .join(" ") .toLocaleLowerCase("pt-BR"); return texto.includes(busca); }); mostrarHistorico(resultado); }); } // Carrega as opções e o histórico assim que a página abre. Promise.all([carregarInsumos(), carregarHistorico()]).catch(function (erro) { console.error(erro); alert(erro.message); }); });
